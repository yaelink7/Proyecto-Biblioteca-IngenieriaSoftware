    private void actualizarDias() {
        String mes = (String) PresMes.getSelectedItem();
        String mesd = (String) DevMes.getSelectedItem();
        if (mes == null || mesd == null) return;
        int diasMaximos = 31;
        int diasMaximosD = 31;
        switch (mes) {
            case "Abril":
            case "Junio":
            case "Septiembre":
            case "Noviembre":
                diasMaximos = 30;
                diasMaximosD = 30;
                break;
            case "Febrero":
                    diasMaximos = 28;
                    diasMaximosD = 28;
                break;
            default:
                diasMaximos = 31;
                diasMaximosD = 31;
                break;
        }
        String diaSeleccionado = (String) PresDia.getSelectedItem();
        String diaSeleccionadoD = (String) DevDia.getSelectedItem();
        javax.swing.DefaultComboBoxModel<String> modeloDias = new javax.swing.DefaultComboBoxModel<>();
        javax.swing.DefaultComboBoxModel<String> modeloDiasD = new javax.swing.DefaultComboBoxModel<>();
        for (int i = 1; i <= diasMaximos; i++) {
            modeloDias.addElement(String.valueOf(i));
        }
        for (int i = 1; i <= diasMaximosD; i++) {
            modeloDiasD.addElement(String.valueOf(i));
        }
        PresDia.setModel(modeloDias);
        DevDia.setModel(modeloDiasD);
        if (diaSeleccionado != null||diaSeleccionadoD != null) {
            int diaAnt = Integer.parseInt(diaSeleccionado);
            if (diaAnt <= diasMaximos) {
                PresDia.setSelectedItem(diaSeleccionado);
            }
            int diaAntD = Integer.parseInt(diaSeleccionadoD);
            if (diaAntD <= diasMaximosD) {
                DevDia.setSelectedItem(diaSeleccionadoD);
            }

      }
    }
    
    private void cargarLibrosBox() {
        LibroDao libroDao = new LibroDao("datosLibros.csv");
        TituLibro.removeAllItems();
        try {
            List<Libro> listaLibros = libroDao.obtenerLibros();
            for (Libro libro : listaLibros) {
                if (libro.getExistencias() > 0) { 
                    TituLibro.addItem(libro.getTitulo());
                }
            }
        } catch (Exception e) {
            JOptionPane.showMessageDialog(this, "Error al cargar libros: " + e.getMessage());
        }
    }
    
        private void cargarUsuariosBox() {
        DAOBiblia.UsuarioDao usuarioDao = new DAOBiblia.UsuarioDao("datosUsuario.csv");
        NomUsuario.removeAllItems();

        try {
            List<ClasesBiblia.Usuario> lista = usuarioDao.obtenerUsuarios();
            for (ClasesBiblia.Usuario u : lista) {
                NomUsuario.addItem(u.getId() + " - " + u.getNombre() + " " + u.getApellido());
            }
        } catch (Exception e) {
            JOptionPane.showMessageDialog(this, "Error al cargar usuarios: " + e.getMessage());
        }
    }
    
    private String obtenerNumeroMes(String mes) {
    switch (mes) {
        case "Enero": return "01";
        case "Febrero": return "02";
        case "Marzo": return "03";
        case "Abril": return "04";
        case "Mayo": return "05";
        case "Junio": return "06";
        case "Julio": return "07";
        case "Agosto": return "08";
        case "Septiembre": return "09";
        case "Octubre": return "10";
        case "Noviembre": return "11";
        case "Diciembre": return "12";
        default: return "01";
    }
}

    private void PrestamoActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_PrestamoActionPerformed
        // TODO add your handling code here:
        String tituloSeleccionado = (String) TituLibro.getSelectedItem();
        String usuarioSeleccionado = (String) NomUsuario.getSelectedItem();
        String diaP = (String) PresDia.getSelectedItem();
        String mesP = obtenerNumeroMes((String) PresMes.getSelectedItem());
        String anoP = (String) PresAno.getSelectedItem();
        String fechaPrestamoSQL = anoP + "-" + mesP + "-" + diaP;

        String diaD = (String) DevDia.getSelectedItem();
        String mesD = obtenerNumeroMes((String) DevMes.getSelectedItem());
        String anoD = (String) DevAno.getSelectedItem();
        String fechaDevolucionSQL = anoD + "-" + mesD + "-" + diaD;
        
        if (tituloSeleccionado == null || usuarioSeleccionado == null) {
         JOptionPane.showMessageDialog(this, "Selecciona un libro y un usuario.");
         return;
        }

        try {
            String[] partesUsuario = usuarioSeleccionado.split(" - ");
            int idUsuario = Integer.parseInt(partesUsuario[0]);
            DAOBiblia.LibroDao libroDao = new DAOBiblia.LibroDao("datosLibros.csv");
            ClasesBiblia.Libro libro = libroDao.buscarTitulo(tituloSeleccionado);

            if (libro != null) {
                if (libro.getExistencias() > 0) {
                    DAOBiblia.PrestamoDao prestamoDao = new DAOBiblia.PrestamoDao();
                    boolean exito = prestamoDao.registrarPrestamo(libro.getIdLibro(), idUsuario, fechaPrestamoSQL, fechaDevolucionSQL);
                    if (exito) {
                        JOptionPane.showMessageDialog(this, "Préstamo realizado con éxito.\nQuedan " + (libro.getExistencias() - 1) + " ejemplares.");

                    } else {
                        JOptionPane.showMessageDialog(this, "Error al guardar en base de datos.", "Error", JOptionPane.ERROR_MESSAGE);
                    }
                } else {
                    JOptionPane.showMessageDialog(this, "No hay existencias disponibles de este libro.", "Agotado", JOptionPane.WARNING_MESSAGE);
                }
            } else {
                JOptionPane.showMessageDialog(this, "Error: Libro no encontrado en BD.", "Error", JOptionPane.ERROR_MESSAGE);
            }

        } catch (Exception e) {
            JOptionPane.showMessageDialog(this, "Error: " + e.getMessage());
        }
    }//GEN-LAST:event_PrestamoActionPerformed
