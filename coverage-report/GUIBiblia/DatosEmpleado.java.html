    private void InicializarComponentesUsuario(){
        tableModelEmpleado = new DefaultTableModel(new Object[]{"Id","Nombre","Apellidos","Calle","Colonia","Numero","Codigo Postal","Correo","TipoEmpleado"},0 );
        TablaEmpleao.setModel(tableModelEmpleado);
        ajusteColumnas();
        empleaoDao = new EmpleadoDao("datosEmpleado.csv");
        listaEmpleados = new ArrayList<>();
        cargarDatosTablaEmpleado();
    }
    
    private void cargarDatosTablaEmpleado(){
        try {
            //listaEmpleados = empleaoDao.cargarEmpleadoCSV();
            listaEmpleados = empleaoDao.obtenerEmpleados();
            tableModelEmpleado.setRowCount(0);
            for(Empleado a:listaEmpleados){
                tableModelEmpleado.addRow(new Object[]{a.getId(),a.getNombre(),a.getApellido(),a.getCalle(),a.getColonia(),a.getNumero(),a.getCodigoPostal(),a.getCorreo(),a.getTipoEmpleado()});
            }
        } catch (Exception e) {
            JOptionPane.showMessageDialog(this, "Error al cargar los datos)"+e.getMessage(),"Error",JOptionPane.ERROR_MESSAGE);
        }
    }

    private void EliminarActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_EliminarActionPerformed
        // TODO add your handling code here:
        int filaSeleccionada = TablaEmpleao.getSelectedRow();
        if (filaSeleccionada >= 0) {
            String nombreEmpleado = (String) TablaEmpleao.getValueAt(filaSeleccionada, 1);
            int confirmar = JOptionPane.showConfirmDialog(this, "¿Estás seguro de Ejecutar al Empleado " + nombreEmpleado + "?\nEsta acción no se puede deshacer, tendras que contratar otro random.",
                "Confirmar Eliminación",JOptionPane.YES_NO_OPTION);

            if (confirmar == JOptionPane.YES_OPTION) {
                try {
                    Empleado empleadoAEliminar = listaEmpleados.get(filaSeleccionada);
                    empleaoDao.eliminarEmpleado(empleadoAEliminar.getId());
                    listaEmpleados.remove(filaSeleccionada);
                    empleaoDao.guardarEmpleadoCSV(listaEmpleados);
                    cargarDatosTablaEmpleado();
                    JOptionPane.showMessageDialog(this, "Hemos ejecutado correctamente al Empleado Patron ?\n y sin compensacion", "Éxito", JOptionPane.INFORMATION_MESSAGE);
                } catch (IOException e) {
                    JOptionPane.showMessageDialog(this, "Error con el CSV" + e.getMessage());
                } catch (Exception e) {
                    JOptionPane.showMessageDialog(this, "Error al eliminar: " + e.getMessage());
                }
            }
        } else {
            JOptionPane.showMessageDialog(this, "Por favor, selecciona un usuario de la tabla para eliminar.", "Aviso", JOptionPane.WARNING_MESSAGE);
        }
    }//GEN-LAST:event_EliminarActionPerformed

    private void EditarActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_EditarActionPerformed
        // TODO add your handling code here:
        int filaSeleccionada = TablaEmpleao.getSelectedRow();
        if (filaSeleccionada >= 0) {
            Empleado empleadoSeleccionado = listaEmpleados.get(filaSeleccionada);
            GUIEmpleado ventanaEdicion = new GUIEmpleado();
            ventanaEdicion.ModoEdicion(empleadoSeleccionado);
            ventanaEdicion.setVisible(true);
            this.dispose();
        } else {
            JOptionPane.showMessageDialog(this, "Selecciona minimo un usuario de la tabla para poder editar.", "Aviso", JOptionPane.WARNING_MESSAGE);
        }
    }//GEN-LAST:event_EditarActionPerformed
